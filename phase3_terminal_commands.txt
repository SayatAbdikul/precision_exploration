# 1. Set up the existing environment and a new log directory.
set -Eeuo pipefail
cd -- "$(dirname -- "${BASH_SOURCE[0]:-phase3_terminal_commands.txt}")"
export PATH="/usr/local/cuda/bin:$PATH"
export OMP_NUM_THREADS=4
export PYTHONUNBUFFERED=1
export CUDAHOSTCXX="${CUDAHOSTCXX:-g++-14}"
export PHASE3_CHECK_DIR="artifacts/phase3/manual-checks/$(date -u +%Y%m%dT%H%M%SZ)-$$"
mkdir -p "$PHASE3_CHECK_DIR"
printf 'Evidence directory: %s\n' "$PHASE3_CHECK_DIR"

# Preserve the failing command's exit status even when output goes through tee.
# A failure stops this script. Retain its log; do not treat skipped/failed checks
# as a pass or change the frozen definitions merely to make a check succeed.
run_check() {
    local check_name="$1"
    shift
    printf '\nRunning %s\n' "$check_name"
    if "$@" 2>&1 | tee "$PHASE3_CHECK_DIR/$check_name.log"; then
        printf '%s PASS\n' "$check_name" >> "$PHASE3_CHECK_DIR/check-status.txt"
    else
        local check_status=$?
        printf '%s FAIL exit=%s\n' "$check_name" "$check_status" >> "$PHASE3_CHECK_DIR/check-status.txt"
        printf 'Stopped. Review %s/%s.log\n' "$PHASE3_CHECK_DIR" "$check_name" >&2
        return "$check_status"
    fi
}

run_check source-state .venv/bin/python - <<'PY'
import importlib.metadata
import json
import os
import subprocess
from pathlib import Path
import numpy, pytest, torch, torchvision, ultralytics, pycocotools.coco
from public.inference.conformance_job import source_identity
from tools.phase3.common import campaign, digest
from tools.phase3.screening import pipeline_identity

record = {
    "git_commit": subprocess.check_output(["git", "rev-parse", "HEAD"], text=True).strip(),
    "git_status": subprocess.check_output(["git", "status", "--short"], text=True),
    "source_sha256": source_identity(),
    "pipeline_sha256": pipeline_identity(),
    "campaign_sha256": digest(campaign()),
    "versions": {name: importlib.metadata.version(name) for name in
                 ("numpy", "pytest", "torch", "torchvision", "ultralytics", "pycocotools")},
}
destination = Path(os.environ["PHASE3_CHECK_DIR"]) / "source-state.json"
destination.write_text(json.dumps(record, indent=2) + "\n")
print(json.dumps(record, indent=2))
PY

run_check environment .venv/bin/python -m tools.setup.check_phase2_environment \
    --output "$PHASE3_CHECK_DIR/environment.json"
# The environment command is a readiness report, not a Phase 3 completion gate.
# Optional hardware tools or the optional EfficientNet workload can be absent.

# This includes unit, integration, regression and native/rational/INT64

# 3. Repeat the backend-dependent conformance suites on the actual GPU.
# The rational suite exercises every accepted operand format; the width suite
# checks the Phase 3 INT64 path, saturation, ordering and wide stored biases.
# These tests need nvcc, the CUDA-compatible host compiler and a visible GPU.
# They can contend with an existing screen, so test-time timings are not
# standalone inference benchmarks. They do not stop or replace that screen.

run_check nvcc "${NVCC:-nvcc}" --version
run_check cuda-host-compiler "$CUDAHOSTCXX" --version
run_check pytest-cuda env PRECISION_TEST_BACKEND=cuda \
    .venv/bin/python -m pytest \
    tests/conformance/test_native_engine.py \
    tests/conformance/test_rational_engine.py \
    tests/conformance/test_phase3_integer_width.py \
    -q -ra --junitxml="$PHASE3_CHECK_DIR/pytest-cuda.xml"

# To include this section, use this invocation instead of the one at the top:
#   PHASE3_RUN_EXPERIMENTS=1 bash phase3_terminal_commands.txt

if [[ "${PHASE3_RUN_EXPERIMENTS:-0}" == "1" ]]; then
    run_check integer-continuation .venv/bin/python -m tools.run.phase3_continue_integer \

        --model resnet18 --format posit6_es1 --scope pilot --images 8 \

    # Analyze every completed current-engine canonical pilot/screen. Historical
    # source identities and incomplete jobs are recorded as skipped, not reused.
    run_check completed-run-analysis .venv/bin/python - <<'PY'
import subprocess
import sys
from public.inference.conformance_job import source_identity
from tools.phase3.common import ROOT, read
from tools.phase3.evidence import verify_complete

source = source_identity()
for path in sorted((ROOT / "artifacts/phase3/runs").glob("*/summary.json")):
    summary = read(path)
    if summary.get("status") != "completed" or summary.get("source_sha256") != source:
        print("SKIP non-current/incomplete summary:", path, flush=True)
        continue
    verify_complete(path, current_execution=True)
    commands = [("tools.analysis.phase3_results", "--summary"),
                ("tools.analysis.phase3_diagnostics", "--summary")]
    if summary["scope"] == "pilot":
        commands.append(("tools.analysis.phase3_shape_check", "--pilot"))
    for module, flag in commands:
        subprocess.run([sys.executable, "-m", module, flag, str(path)], check=True)
PY
    run_check phase3-status-after-runs .venv/bin/python -m tools.analysis.phase3_status
    run_check phase3-gates-after-runs .venv/bin/python -m tools.analysis.phase3_gate_inventory
